// Uniformizeaza comanda: minuscule si un singur spatiu intre cuvinte.
void normalizeCommand(char* text) {
    size_t readIndex = 0;
    size_t writeIndex = 0;
    bool previousSpace = false;

    while (isspace((unsigned char)text[readIndex])) readIndex++;

    while (text[readIndex] != '\0') {
        char c = (char)tolower((unsigned char)text[readIndex++]);
        if (isspace((unsigned char)c)) {
            if (writeIndex > 0 && !previousSpace) {
                text[writeIndex++] = ' ';
                previousSpace = true;
            }
        }
        else {
            text[writeIndex++] = c;
            previousSpace = false;
        }
    }

    if (writeIndex > 0 && text[writeIndex - 1] == ' ') writeIndex--;
    text[writeIndex] = '\0';
}

// Interpreteaza comanda primita si apeleaza actiunea motorului potrivita.
void processSerialCommand(char* command) {
    normalizeCommand(command);
    if (command[0] == '\0') return;

    printf("\n[COMANDA] %s\n", command);
    char action[16] = "";
    char subcommand[16] = "";
    char valueText[16] = "";
    char extra[2] = "";
    // Separa cuvintele si detecteaza argumente suplimentare nepermise.
    int count = sscanf(command, "%15s %15s %15s %1s", action, subcommand, valueText, extra);

    if (count == 3 && strcmp(action, "motor") == 0 && strcmp(subcommand, "set") == 0) {
        char* endPtr;
        // Conversia permite verificarea stricta a valorii numerice introduse.
        long value = strtol(valueText, &endPtr, 10);
        if (endPtr == valueText || *endPtr != '\0' || value < MIN_POWER || value > MAX_POWER) {
            printf("[EROARE] Puterea trebuie sa fie intre -100 si 100.\n");
            displayError("Valoare invalida");
            return;
        }
        setMotorPower((int)value);
        printf("[OK] Puterea motorului a fost setata.\n");
        reportChange();
    }
    else if (strcmp(command, "motor stop") == 0) {
        stopMotor();
        printf("[OK] Motor oprit!\n");
        reportChange();
    }
    else if (strcmp(command, "motor max") == 0) {
        maxMotor();
        printf("[OK] Putere maxima!\n");
        reportChange();
    }
    else if (strcmp(command, "motor inc") == 0) {
        increaseMotorPower();
        reportChange();
    }
    else if (strcmp(command, "motor dec") == 0) {
        decreaseMotorPower();
        reportChange();
    }
    else if (strcmp(command, "status") == 0) reportChange();
    else if (strcmp(command, "help") == 0) showMenu();
    else {
        printf("[EROARE] Comanda necunoscuta!\n");
        printf("Tasteaza help pentru comenzi.\n");
        displayError("Comanda invalida");
    }
}

const byte ROWS = 4;
const byte COLS = 4;
char keys[ROWS][COLS] = {{'1', '2', '3', 'A'}, {'4', '5', '6', 'B'}, {'7', '8', '9', 'C'}, {'*', '0', '#', 'D'}};
byte rowPins[ROWS] = {22, 23, 24, 25};
byte colPins[COLS] = {26, 27, 28, 29};
// Leaga harta tastelor de pinii fizici ai keypad-ului 4x4.
Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// Debounce-ul elimina detectarea repetata a unei singure apasari.
void initKeypad() {
    keypad.setDebounceTime(30);
}

// Converteste tastele A-D in comenzi directe pentru motor.
void readKeypad() {
    char key = keypad.getKey();
    // Daca nu exista tasta noua, se revine imediat in loop().
    if (key == NO_KEY) return;
    printf("\n[KEYPAD] Tasta: %c\n", key);

    switch (key) {
        case 'A': increaseMotorPower(); reportChange(); break;
        case 'B': decreaseMotorPower(); reportChange(); break;
        case 'C': reverseMotorDirection(); printf("[OK] Directie schimbata!\n"); reportChange(); break;
        case 'D': stopMotor(); printf("[STOP] Motor oprit!\n"); reportChange(); break;
        default:
            printf("[EROARE] Tasta neutilizata: %c\n", key);
            displayError("Foloseste A-D");
            break;
    }
}

// Obiectul care controleaza LCD-ul conectat prin I2C.
hd44780_I2Cexp lcd;
bool lcdAvailable = false;

// Initializeaza I2C si verifica disponibilitatea LCD-ului.
void initLcd() {
    Wire.begin();
    // LCD-ul este configurat pentru 16 coloane si 2 randuri.
    int result = lcd.begin(16, 2);

    if (result != 0) {
        printf("[EROARE LCD] Cod: %d\n", result);
        lcdAvailable = false;
        return;
    }

    lcdAvailable = true;
    lcd.backlight();
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("LABORATOR 4.2");
    lcd.setCursor(0, 1);
    lcd.print("MOTOR DC L293D");
    delay(1200);
}

// Afiseaza un mesaj de eroare numai cand LCD-ul a fost initializat.
void displayError(const char* message) {
    if (!lcdAvailable) return;
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("EROARE!");
    lcd.setCursor(0, 1);
    lcd.print(message);
}

// Actualizeaza LCD-ul cu directia efectiva si puterea curenta.
void displayMotorState() {
    if (!lcdAvailable) return;
    lcd.clear();
    lcd.setCursor(0, 0);

    if (motorPower > 0) lcd.print("DIR: FORWARD");
    else if (motorPower < 0) lcd.print("DIR: REVERSE");
    else lcd.print("MOTOR: STOP");

    lcd.setCursor(0, 1);
    lcd.print("POWER: ");
    lcd.print(abs(motorPower));
    lcd.print("%");
}

// Ruleaza o singura data pentru initializarea tuturor modulelor.
void setup() {
    // Viteza trebuie sa fie aceeasi cu monitor_speed din platformio.ini.
    Serial.begin(9600);
    initStdio();
    initMotorDriver();
    initLcd();
    initKeypad();

    stopMotor();
    displayMotorState();

    printf("\nSistem initializat!\n");
    printf("Motor initial: STOP\n");
    showMenu();
    reportChange();
    printf("\nIntrodu o comanda: ");
}

void loop() {
    readSerialCommand();
    readKeypad();
    periodicReport();
}

// Impune limitele de siguranta ale puterii motorului.
int clampPower(int power) {
    return constrain(power, MIN_POWER, MAX_POWER);
}

// Salveaza puterea, actualizeaza directia memorata si comanda driverul.
void setMotorPower(int power) {
    power = clampPower(power);
    motorPower = power;

    if (power > 0) {
        motorDirection = 1;
    }
    else if (power < 0) {
        motorDirection = -1;
    }

    writeMotorHardware(motorPower);
}

// Opreste motorul pastrand ultima directie folosita.
void stopMotor() {
    setMotorPower(0);
}

// Aplica puterea maxima pe directia memorata.
void maxMotor() {
    setMotorPower(motorDirection * MAX_POWER);
}

// Creste magnitudinea puterii in pasi de 10%.
void increaseMotorPower() {
    int currentMagnitude = abs(motorPower);

    if (currentMagnitude >= MAX_POWER) {
        printf("[LIMITA] Puterea maxima a fost atinsa!\n");
        return;
    }

    int newMagnitude = min(currentMagnitude + POWER_STEP, MAX_POWER);
    setMotorPower(motorDirection * newMagnitude);
}

// Reduce magnitudinea puterii spre zero in pasi de 10%.
void decreaseMotorPower() {
    int currentMagnitude = abs(motorPower);

    if (currentMagnitude == 0) {
        printf("[LIMITA] Motorul este deja oprit!\n");
        return;
    }

    int newMagnitude = max(currentMagnitude - POWER_STEP, 0);
    setMotorPower(motorDirection * newMagnitude);
}

// Inverseaza sensul si mentine aceeasi putere daca motorul este activ.
void reverseMotorDirection() {
    motorDirection *= -1;

    if (motorPower != 0) {
        setMotorPower(motorDirection * abs(motorPower));
    }
}

// Configureaza pinii driverului L293D si porneste cu motorul oprit.
void initMotorDriver() {
    digitalWrite(MOTOR_ENA, LOW);
    digitalWrite(MOTOR_IN1, LOW);
    digitalWrite(MOTOR_IN2, LOW);

    pinMode(MOTOR_ENA, OUTPUT);
    pinMode(MOTOR_IN1, OUTPUT);
    pinMode(MOTOR_IN2, OUTPUT);

    // Dezactiveaza PWM-ul inaintea unei eventuale schimbari de directie.
    analogWrite(MOTOR_ENA, 0);
}

// Convertește procentul de putere in PWM si selecteaza sensul motorului.
void writeMotorHardware(int power) {
    // Conversie din intervalul procentual 0..100 in intervalul PWM 0..255.
    int pwmValue = map(abs(power), 0, 100, 0, 255);

    analogWrite(MOTOR_ENA, 0);

    if (power > 0) {
        digitalWrite(MOTOR_IN1, HIGH);
        digitalWrite(MOTOR_IN2, LOW);
    }
    else if (power < 0) {
        digitalWrite(MOTOR_IN1, LOW);
        digitalWrite(MOTOR_IN2, HIGH);
    }
    else {
        digitalWrite(MOTOR_IN1, LOW);
        digitalWrite(MOTOR_IN2, LOW);
    }

    analogWrite(MOTOR_ENA, pwmValue);
}

// Afiseaza starea motorului in Serial Monitor si apoi pe LCD.
void reportMotorState() {
    const char* direction;
    if (motorPower > 0) direction = "FORWARD";
    else if (motorPower < 0) direction = "REVERSE";
    else direction = "STOP";

    int pwmValue = map(abs(motorPower), 0, 100, 0, 255);
    printf("\n------- STARE MOTOR -------\n");
    printf("Directie: %s\n", direction);
    printf("Putere setata: %d%%\n", motorPower);
    printf("PWM: %d / 255\n", pwmValue);
    printf("Directie memorata: %s\n", motorDirection == 1 ? "FORWARD" : "REVERSE");
    printf("---------------------------\n");
    displayMotorState();
}

// Raporteaza imediat schimbarea si reporneste cronometrul de raportare.
void reportChange() {
    reportMotorState();
    lastReport = millis();
}

// Tipareste comenzile disponibile pentru Serial Monitor si keypad.
void showMenu() {
    printf("\n================================\n");
    printf(" LABORATOR 4.2 - MOTOR L293D\n");
    printf("================================\n");
    printf("COMENZI SERIAL:\n");
    printf(" motor set N (-100..100)\n motor stop\n motor max\n motor inc\n motor dec\n status\n help\n");
    printf("\nCOMENZI KEYPAD:\n");
    printf(" A - Creste puterea +10%%\n B - Reduce puterea -10%%\n C - Schimba directia\n D - STOP imediat\n");
    printf("================================\n");
}

// Raporteaza fara delay(), dupa expirarea intervalului configurat.
void periodicReport() {
    unsigned long now = millis();
    if (now - lastReport >= REPORT_INTERVAL) {
        reportMotorState();
        lastReport = now;
    }
}

// Bufferul retine o singura comanda, iar dimensiunea limiteaza intrarea.
char serialBuffer[48];
size_t serialIndex = 0;
bool inputOverflow = false;

// Citeste serial neblocant pentru ca keypad-ul si raportarea sa ramana active.
void readSerialCommand() {
    while (Serial.available() > 0) {
        char c = (char)Serial.read();
        // Enter finalizeaza comanda acumulata in buffer.
        if (c == '\r' || c == '\n') {
            if (serialIndex == 0 && !inputOverflow) continue;
            printf("\n");
            if (inputOverflow) {
                printf("[EROARE] Comanda prea lunga!\n");
                displayError("Text prea lung");
            }
            else {
                serialBuffer[serialIndex] = '\0';
                processSerialCommand(serialBuffer);
            }
            serialIndex = 0;
            inputOverflow = false;
            printf("\nIntrodu o comanda: ");
        }
        // Backspace sterge ultimul caracter din buffer si din afisarea seriala.
        else if (c == '\b' || c == 127) {
            if (serialIndex > 0) {
                serialIndex--;
                Serial.print("\b \b");
            }
        }
        else if (isprint((unsigned char)c)) {
            // Echo pentru ca utilizatorul sa vada caracterul tastat.
            Serial.write(c);
            if (!inputOverflow) {
                if (serialIndex < sizeof(serialBuffer) - 1) serialBuffer[serialIndex++] = c;
                else inputOverflow = true;
            }
        }
    }
}

// Fluxul stdout este redirectionat spre Serial Monitor.
static FILE serialOutput;

// Trimite fiecare caracter rezultat din printf() prin interfata seriala.
int serialPutChar(char c, FILE* stream) {
    (void)stream;

    if (c == '\n') {
        Serial.write('\r');
    }

    Serial.write(c);
    return 0;
}

// Leaga stdout de serialPutChar pentru a permite folosirea printf().
void initStdio() {
    fdev_setup_stream(&serialOutput, serialPutChar, NULL, _FDEV_SETUP_WRITE);
    stdout = &serialOutput;
}
